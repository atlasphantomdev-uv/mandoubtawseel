Only config-sourceable facts: url, logo, areaServed. */}} {{- $business := dict "@type" "LocalBusiness" "@id" $businessID "url" $siteURL "logo" $logo "areaServed" (dict "@type" "Country" "name" "AdministrativeArea" "name" .) -}} {{- end -}} {{- $areaServed = $areaServed | append (dict "@type" "City" "name" "العين") -}} {{- $service := dict "@type" "Service" "@id" (printf "%s#service" $siteURL) "serviceType" "Courier and delivery service" "provider" (dict "@id" $businessID) "areaServed" $areaServed "availableLanguage" (slice "ar" "en") -}} {{- $webpage := dict "@type" "WebPage" $webpage = merge $webpage (dict "description" .) }}{{ end -}} {{- if not $page.IsHome }}{{ $webpage = merge $webpage (dict "about" (dict "@id" $businessID)) }}{{ end -}} {{- $graph := slice $website $business $service $webpage -}} {{/* Article node for blog posts. */}} {{- if eq $page.Type "blog" -}} {{- $post := dict "@type" "BlogPosting" "@id" (printf "%s#article" $page.Permalink) $post (dict "datePublished" ($page.Date.Format "2006-01-02T15:04:05Z07:00")) }}{{ end -}} {{- if not $page.Lastmod.IsZero }}{{ $post = merge $post (dict "dateModified" ($page.Lastmod.Format "2006-01-02T15:04:05Z07:00")) }}{{ end -}} {{- with $summary }}{{ $post = merge $post (dict "description" .) }}{{ end -}} {{- with partial "functions/get_featured_image" $page }}{{ $post = merge $post (dict "image" (slice .Permalink)) }}{{ end -}} {{- $graph = $graph | append $post -}} {{- end -}} {{/* BreadcrumbList on non-home pages. */}} {{- if not $page.IsHome -}} {{- $items := slice -}} {{- $items = $items | append (dict "@type" "ListItem" "position" $position "name" ($home_title | plainify) "item" (dict "@id" .Permalink "name" ($home_title | plainify))) -}} {{- $position = add $position 1 -}} {{- end -}} {{- range $page.Ancestors.Reverse -}} {{- if not .IsHome -}} {{- $items = $items | append (dict "@type" "ListItem" "position" $position "name" (.LinkTitle | plainify) "item" (dict
